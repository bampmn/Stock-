"use strict";


const KEY = "management-os-standalone-v1";

const PREVIOUS = KEY + "-previous";

const $ = id => document.getElementById(id);

const esc = value => String(value ?? "").replace(/[&<>"']/g, c =>

  ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));

const money = n => new Intl.NumberFormat("th-TH", {

  style:"currency", currency:"THB"

}).format(n);

const number = n => new Intl.NumberFormat("th-TH", {maximumFractionDigits:4}).format(n);

const today = () => {

  const d = new Date();

  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;

};

const formatDate = d => d ? new Date(d+"T12:00:00").toLocaleDateString("th-TH") : "ไม่ทราบวันที่";

const uid = prefix => prefix + "-" + (globalThis.crypto?.randomUUID?.() ||

  Date.now().toString(36)+Math.random().toString(36).slice(2));

const empty = () => ({products:[],suppliers:[],orders:[],lots:[],usages:[]});

const clone = value => JSON.parse(JSON.stringify(value));


let state = empty(), revision = 0, savedAt = "", storedRaw = null;

let view = "inventory", mode = "", editing = "", toastTimer;

let storageBlocked = false;

const views = {

  inventory:"คลังสินค้า",

  purchases:"รายการซื้อ",

  usage:"เบิกใช้งาน",

  suppliers:"ร้านค้า"

};


function notify(message) {

  $("toast").textContent = message;

  $("toast").classList.remove("hidden");

  clearTimeout(toastTimer);

  toastTimer = setTimeout(() => $("toast").classList.add("hidden"), 6500);

}


function validDate(value, optional=false) {

  if (optional && value==="") return true;

  return typeof value==="string" && /^\d{4}-\d{2}-\d{2}$/.test(value) &&

    Number.isFinite(Date.parse(value)) &&

    new Date(value).toISOString().slice(0,10)===value;

}


function safeURL(value) {

  const text = String(value || "").trim();

  if (!text) return "";

  const url = new URL(text);

  if (!["https:","http:"].includes(url.protocol)) throw Error("ลิงก์ต้องเป็น http:// หรือ https://");

  return url.href;

}


/* Accepts backups from this page and the previous Management OS application. */

function validate(s) {

  if (!s || typeof s!=="object") throw Error("ไฟล์ข้อมูลไม่ถูกต้อง");

  for (const key of ["products","suppliers","orders","lots","usages"]) {

    if (!Array.isArray(s[key])) throw Error("ไฟล์ไม่มีตาราง "+key);

    const ids = new Set();

    for (const row of s[key]) {

      if (!row || typeof row.id!=="string" || !row.id || ids.has(row.id))

        throw Error("รหัสข้อมูลว่างหรือซ้ำใน "+key);

      ids.add(row.id);

    }

  }

  const string = (v, max=4000) => typeof v==="string" && v.length<=max;

  const finite = (v, positive=false) => typeof v==="number" && Number.isFinite(v) &&

    (positive ? v>0 : v>=0) && v<=1e12;

  const products = new Set(s.products.map(p=>p.id));

  const suppliers = new Set(s.suppliers.map(p=>p.id));

  const orders = new Set(s.orders.map(p=>p.id));

  const lots = new Map(s.lots.map(p=>[p.id,p]));


  for (const p of s.products)

    if (!string(p.name)||!p.name.trim()||!string(p.variant)||!string(p.unit)||!p.unit.trim()||!string(p.location))

      throw Error("ข้อมูลสินค้าไม่ถูกต้อง");


  for (const p of s.suppliers) {

    if (!string(p.name)||!p.name.trim()||!string(p.url)) throw Error("ข้อมูลร้านค้าไม่ถูกต้อง");

    safeURL(p.url);

  }


  for (const o of s.orders) {

    if (!suppliers.has(o.supplierId)||!validDate(o.date,true)||!finite(o.total)||

        !string(o.dateType)||!string(o.source)||!string(o.url)) throw Error("ข้อมูลคำสั่งซื้อไม่ถูกต้อง");

    safeURL(o.url);

  }


  for (const l of s.lots)

    if (!products.has(l.productId)||!orders.has(l.orderId)||!finite(l.qty,true)||

        !finite(l.cost)||typeof l.estimated!=="boolean"||

        (l.pending!==undefined&&typeof l.pending!=="boolean")) throw Error("ข้อมูลล็อตไม่ถูกต้อง");


  const used = new Map();

  for (const u of s.usages) {

    if (!lots.has(u.lotId)||!finite(u.qty,true)||!validDate(u.date)||

        !string(u.person)||!string(u.note)) throw Error("ข้อมูลการเบิกไม่ถูกต้อง");

    used.set(u.lotId,(used.get(u.lotId)||0)+u.qty);

  }

  for (const [id,q] of used)

    if (lots.get(id).pending||q>lots.get(id).qty+1e-8) throw Error("ยอดเบิกเกินสต็อกหรือเบิกจากล็อตรอรับ");

  return s;

}


function load() {

  try {

    storedRaw = localStorage.getItem(KEY);

    if (storedRaw) {

      const doc = JSON.parse(storedRaw);

      state = validate(doc.state);

      revision = doc.revision || 0;

      savedAt = doc.savedAt || "";

    }

  } catch (e) {

    storageBlocked = true;

    notify("อ่านข้อมูลไม่ได้: "+e.message+" — ยังไม่เขียนทับข้อมูลเดิม");

  }

}


function commit(next) {

  if (storageBlocked) throw Error("พื้นที่เก็บข้อมูลมีปัญหา กรุณาสำรองและตรวจข้อมูลก่อน");

  validate(next);

  if (localStorage.getItem(KEY)!==storedRaw)

    throw Error("ข้อมูลถูกเปลี่ยนจากแท็บอื่น กรุณารีเฟรชก่อนบันทึก");


  const doc = {version:1,state:next,revision:revision+1,savedAt:new Date().toISOString()};

  const raw = JSON.stringify(doc);

  /* Keep the previous successful state before replacing the current one. */

  if (storedRaw) localStorage.setItem(PREVIOUS,storedRaw);

  localStorage.setItem(KEY,raw);

  storedRaw=raw; state=next; revision=doc.revision; savedAt=doc.savedAt;

  render();

}


const product = id => state.products.find(p=>p.id===id);

const supplier = id => state.suppliers.find(p=>p.id===id);

const order = id => state.orders.find(p=>p.id===id);

const remaining = l => l.pending ? 0 : Math.max(0,

  Math.round((l.qty-state.usages.filter(u=>u.lotId===l.id).reduce((sum,u)=>sum+u.qty,0))*1e8)/1e8);


function stock(p) {

  const lots=state.lots.filter(l=>l.productId===p.id);

  return {

    qty:lots.reduce((s,l)=>s+remaining(l),0),

    value:lots.reduce((s,l)=>s+remaining(l)*l.cost/l.qty,0),

    pending:lots.filter(l=>l.pending).reduce((s,l)=>s+l.qty,0)

  };

}


const button = (action,id,text) =>

  `<button class="link whitespace-nowrap mr-3" data-action="${action}" data-id="${esc(id)}">${esc(text)}</button>`;

const external = url => url ? `<a class="link" href="${esc(safeURL(url))}" target="_blank" rel="noopener noreferrer">เปิดลิงก์ ↗</a>` : "—";

const table = (headers,rows) =>

  `<table class="w-full text-sm"><thead><tr>${headers.map(h=>`<th>${h}</th>`).join("")}</tr></thead><tbody>${

    rows.length ? rows.join("") : `<tr><td colspan="${headers.length}" class="text-center py-16 text-slate-400">ยังไม่มีรายการที่ตรงกัน</td></tr>`

  }</tbody></table>`;


function render() {

  $("title").textContent=views[view];

  $("breadcrumb").textContent=views[view];

  $("tableTitle").textContent=views[view];

  $("addSupplier").classList.toggle("hidden",view!=="suppliers");

  $("nav").innerHTML=Object.entries(views).map(([key,label])=>

    `<button data-view="${key}" class="text-left whitespace-nowrap px-4 py-3 rounded-xl ${view===key?"nav-active":"text-slate-500"}">${label}</button>`).join("");


  const values=state.products.map(stock);

  const cards=[

    ["รายการสินค้า",number(state.products.length)],

    ["มูลค่าสต็อกพร้อมใช้",money(values.reduce((s,p)=>s+p.value,0))],

    ["ล็อตรอรับ",number(state.lots.filter(l=>l.pending).length)],

    ["ยอดคำสั่งซื้อรวม",money(state.orders.reduce((s,o)=>s+o.total,0))]

  ];

  $("stats").innerHTML=cards.map(([label,value])=>

    `<div class="bg-white border rounded-2xl p-5"><p class="text-sm text-slate-500">${label}</p><p class="text-2xl font-bold mt-3">${value}</p></div>`).join("");


  const q=$("search").value.trim().toLocaleLowerCase();

  const match=(...v)=>v.join(" ").toLocaleLowerCase().includes(q);

  let rows=[];


  if(view==="inventory") {

    rows=state.products.filter(p=>match(p.id,p.name,p.variant,p.location)).map(p=>{

      const s=stock(p);

      return `<tr>

        <td><b>${esc(p.name)}</b><div class="text-slate-400 text-xs mt-1">${esc(p.variant)} · ${esc(p.id)}</div></td>

        <td>${esc(p.location)||"—"}</td>

        <td><b class="${s.qty?"text-emerald-600":"text-slate-400"}">${number(s.qty)}</b> ${esc(p.unit)}

          ${s.pending?`<div class="text-xs text-amber-600 mt-1">รอรับ ${number(s.pending)} ${esc(p.unit)}</div>`:""}</td>

        <td>${money(s.value)}</td>

        <td>${button("product",p.id,"แก้ไข")}${button("receive",p.id,"ซื้อเพิ่ม")}${button("use",p.id,"เบิก")}</td>

      </tr>`;

    });

    $("content").innerHTML=table(["สินค้า / รุ่น","ที่เก็บ","คงเหลือ","มูลค่า","จัดการ"],rows);

  }


  if(view==="purchases") {

    rows=[...state.lots].reverse().filter(l=>{

      const p=product(l.productId),o=order(l.orderId);

      return match(p.name,p.variant,l.id,o.id,o.source,supplier(o.supplierId).name);

    }).map(l=>{

      const p=product(l.productId),o=order(l.orderId);

      return `<tr>

        <td><b>${esc(p.name)}</b><div class="text-xs text-slate-400 mt-1">${esc(p.variant)} · ${esc(l.id)}</div>

          <div class="text-xs text-slate-400 mt-1">${esc(o.source)}</div></td>

        <td>${esc(supplier(o.supplierId).name)}<div>${external(o.url)}</div></td>

        <td>${formatDate(o.date)}<div class="text-xs text-slate-400">${esc(o.dateType)}</div></td>

        <td>${l.pending?'<span class="text-amber-600">รอรับ</span>':'รับแล้ว'} ${number(l.qty)} ${esc(p.unit)}

          <div class="text-xs text-slate-500">พร้อมใช้ ${number(remaining(l))}</div></td>

        <td>${money(l.cost/l.qty)}</td>

        <td>${money(l.cost)}<div class="text-xs text-slate-400">${l.estimated?"ประมาณการ":"ยืนยันแล้ว"}</div></td>

        <td>${l.pending?button("confirm",l.id,"ยืนยันรับครบ"):""}${button("cost",l.id,"แก้ต้นทุน")}</td>

      </tr>`;

    });

    $("content").innerHTML=table(["สินค้า / ล็อต","ร้านค้า","วันที่","จำนวน","ต้นทุน/หน่วย","ต้นทุนล็อต","จัดการ"],rows);

  }


  if(view==="usage") {

    rows=[...state.usages].reverse().filter(u=>{

      const l=state.lots.find(l=>l.id===u.lotId);

      return match(product(l.productId).name,u.person,u.note);

    }).map(u=>{

      const l=state.lots.find(l=>l.id===u.lotId),p=product(l.productId);

      return `<tr><td>${formatDate(u.date)}</td><td><b>${esc(p.name)}</b><div class="text-xs text-slate-400">${esc(p.variant)}</div></td>

      <td>${number(u.qty)} ${esc(p.unit)}</td><td>${money(u.qty*l.cost/l.qty)}</td><td>${esc(u.person)}</td><td>${esc(u.note)}</td></tr>`;

    });

    $("content").innerHTML=table(["วันที่","สินค้า","จำนวนเบิก","ต้นทุน","ผู้เบิก","ใช้งาน / หมายเหตุ"],rows);

  }


  if(view==="suppliers") {

    rows=state.suppliers.filter(s=>match(s.name,s.url)).map(s=>

      `<tr><td><b>${esc(s.name)}</b></td><td>${external(s.url)}</td><td>${state.orders.filter(o=>o.supplierId===s.id).length}</td><td>${button("supplier",s.id,"แก้ไข")}</td></tr>`);

    $("content").innerHTML=table(["ร้านค้า","ลิงก์","จำนวนคำสั่งซื้อ","จัดการ"],rows);

  }

  $("saved").textContent=savedAt

    ? `บันทึกล่าสุด ${new Date(savedAt).toLocaleString("th-TH")} · รุ่นข้อมูล ${revision}`

    : "เริ่มจากเพิ่มสินค้าและร้านค้า หรือนำเข้าไฟล์สำรอง JSON เดิม";

}


function input(label,name,value="",type="text",required=true,full=false) {

  return `<label class="${full?"sm:col-span-2":""}">${esc(label)}

    <input name="${name}" type="${type}" value="${esc(value)}" ${required?"required":""}

    ${type==="number"?'min="0" step="any"':'maxlength="2000"'}></label>`;

}

function select(label,name,options,value="") {

  return `<label class="sm:col-span-2">${esc(label)}<select name="${name}" required>

    <option value="">เลือก${esc(label)}</option>

    ${options.map(([id,label])=>`<option value="${esc(id)}" ${id===value?"selected":""}>${esc(label)}</option>`).join("")}

  </select></label>`;

}


function openForm(action,id="") {

  mode=action;editing=id;$("formError").textContent="";

  let html="",title="";

  if(action==="product") {

    const p=product(id)||{};

    title=id?"แก้ไขสินค้า":"เพิ่มสินค้า";

    html=input("ชื่อสินค้า","name",p.name||"","text",true,true)+

      input("รุ่น / ขนาด","variant",p.variant||"","text",false,true)+

      input("หน่วย","unit",p.unit||"ชิ้น")+input("ที่เก็บ","location",p.location||"","text",false);

  if(action==="supplier") {

    const s=supplier(id)||{};

    html=input("ชื่อร้าน","name",s.name||"","text",true,true)+input("ลิงก์ร้าน","url",s.url||"","url",false,true);

  }

  if(action==="receive") {

$("form").addEventListener("submit",e=>{

  e.preventDefault();

  try {

    const f=Object.fromEntries(new FormData(e.target));

    const next=clone(state);

    const amount=(name,positive=false)=>{

      const value=Number(f[name]);

      if(f[name]===""||!Number.isFinite(value)||(positive?value<=0:value<0)||value>1e9)

        throw Error("กรุณากรอกจำนวนหรือต้นทุนให้ถูกต้อง");

      return value;

    };

    if(mode==="product") {

      const p={id:editing||uid("P"),name:f.name.trim(),variant:f.variant.trim(),unit:f.unit.trim(),location:f.location.trim()};

      const index=next.products.findIndex(x=>x.id===editing);

      if(index>=0) {

        if(next.lots.some(l=>l.productId===editing)&&p.unit!==next.products[index].unit)

          throw Error("สินค้าที่มีประวัติซื้อแล้วไม่สามารถเปลี่ยนหน่วยได้");

        next.products[index]=p;

      } else next.products.push(p);

    }

    if(mode==="supplier") {

      const s={id:editing||uid("S"),name:f.name.trim(),url:safeURL(f.url)};

      const index=next.suppliers.findIndex(x=>x.id===editing);

      if(index>=0) next.suppliers[index]=s; else next.suppliers.push(s);

    }

    if(mode==="receive") {

      const id=uid("O"),pending=f.status==="pending",cost=amount("cost");

      next.orders.push({id,date:f.date,dateType:pending?"บันทึกซื้อ / รอรับ":"รับเข้า",

        supplierId:f.supplierId,total:cost,source:f.source.trim()||"บันทึกด้วยตนเอง",url:safeURL(f.url)});

      next.lots.push({id:uid("L"),productId:f.productId,orderId:id,qty:amount("qty",true),cost,estimated:false,pending});

    }

    if(mode==="use") {

      const l=state.lots.find(l=>l.id===f.lotId),qty=amount("qty",true);

      if(!l||qty>remaining(l)+1e-8) throw Error("จำนวนเบิกเกินสินค้าพร้อมใช้");

      next.usages.push({id:uid("U"),lotId:l.id,qty,date:f.date,person:f.person.trim(),note:f.note.trim()});

    }

    if(mode==="confirm") {

      const l=next.lots.find(l=>l.id===editing);

      if(!l?.pending) throw Error("ล็อตนี้ไม่ได้รอรับสินค้า");

      l.pending=false;

      const o=next.orders.find(o=>o.id===l.orderId);

      o.date=f.date;o.dateType="รับเข้า";

    }

    if(mode==="cost") {

      const l=next.lots.find(l=>l.id===editing);

      l.cost=amount("cost");l.estimated=false;

    }

    commit(next);

    $("dialog").close();

    notify("บันทึกเรียบร้อย");

  } catch(e) {

    $("formError").textContent=e.message;

  }

});

function download(filename,text) {

  const url=URL.createObjectURL(new Blob([text],{type:"application/json"}));

  const a=document.createElement("a");a.href=url;a.download=filename;a.click();

  setTimeout(()=>URL.revokeObjectURL(url),1000);

}

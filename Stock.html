<script>

"use strict";


const $=id=>document.getElementById(id);

const KEY="management-os-workflows-v1";

const PREVIOUS=KEY+"-previous";

const esc=v=>String(v??"").replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));

const copy=v=>JSON.parse(JSON.stringify(v));

const money=v=>new Intl.NumberFormat("th-TH",{style:"currency",currency:"THB"}).format(v);

const num=v=>new Intl.NumberFormat("en",{maximumFractionDigits:4}).format(v);

const today=()=>{

  const d=new Date();

  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;

};

const uid=p=>p+"-"+(globalThis.crypto?.randomUUID?.()||Date.now().toString(36)+Math.random().toString(36).slice(2));

const fmt=d=>d?new Date(d+"T12:00:00").toLocaleDateString():"Unknown date";

const empty=()=>({products:[],suppliers:[],orders:[],lots:[],usages:[],tasks:[]});

const stages=["Backlog","In progress","Review","Done"];

const views={

  overview:["Overview","A clear view of your stock and work."],

  inventory:["Inventory","Know what is available, incoming, and ready to use."],

  purchases:["Purchases","Track each purchase lot and its actual cost."],

  usage:["Usage","Record where your inventory goes."],

  suppliers:["Suppliers","Keep your purchasing contacts in one place."],

  workflows:["Workflows","Plan, assign, and move work through your process."]

};


let state=empty(),view="overview",revision=0,savedAt="",raw=null;

let mode="",editing="",timer,blocked=false;


function notify(message){

  $("toast").textContent=message;

  $("toast").classList.remove("hidden");

  clearTimeout(timer);

  timer=setTimeout(()=>$("toast").classList.add("hidden"),6500);

}

function validDate(d,optional=false){

  if(optional&&d==="")return true;

  return typeof d==="string"&&/^\d{4}-\d{2}-\d{2}$/.test(d)&&

    Number.isFinite(Date.parse(d))&&new Date(d).toISOString().slice(0,10)===d;

}

function safeURL(v){

  if(!v)return "";

  const u=new URL(v);

  if(!["http:","https:"].includes(u.protocol))throw Error("Use an http:// or https:// URL.");

  return u.href;

}

function validate(s){

  if(!s||typeof s!=="object")throw Error("Invalid backup.");

  if(s.tasks===undefined)s.tasks=[];

  for(const key of ["products","suppliers","orders","lots","usages","tasks"]){

    if(!Array.isArray(s[key]))throw Error("Missing table: "+key);

    const ids=new Set();

    for(const row of s[key]){

      if(!row||typeof row.id!=="string"||!row.id||ids.has(row.id))throw Error("Invalid or duplicate ID in "+key);

      ids.add(row.id);

    }

  }

  const text=v=>typeof v==="string"&&v.length<=10000;

  const value=(v,positive=false)=>typeof v==="number"&&Number.isFinite(v)&&(positive?v>0:v>=0)&&v<=1e12;

  const ps=new Set(s.products.map(p=>p.id)),ss=new Set(s.suppliers.map(x=>x.id));

  const os=new Set(s.orders.map(o=>o.id)),ls=new Map(s.lots.map(l=>[l.id,l]));

  for(const p of s.products)

    if(!text(p.name)||!p.name.trim()||!text(p.variant)||!text(p.unit)||!p.unit.trim()||!text(p.location))

      throw Error("Invalid product.");

  for(const x of s.suppliers){

    if(!text(x.name)||!x.name.trim()||!text(x.url))throw Error("Invalid supplier.");

    safeURL(x.url);

  }

  for(const o of s.orders){

    if(!ss.has(o.supplierId)||!validDate(o.date,true)||!value(o.total)||!text(o.source)||!text(o.dateType)||!text(o.url))

      throw Error("Invalid order.");

    safeURL(o.url);

  }

  for(const l of s.lots)

    if(!ps.has(l.productId)||!os.has(l.orderId)||!value(l.qty,true)||!value(l.cost)||

       typeof l.estimated!=="boolean"||(l.pending!==undefined&&typeof l.pending!=="boolean"))

      throw Error("Invalid lot.");

  const used=new Map();

  for(const u of s.usages){

    if(!ls.has(u.lotId)||!value(u.qty,true)||!validDate(u.date)||!text(u.person)||!text(u.note))

      throw Error("Invalid usage record.");

    used.set(u.lotId,(used.get(u.lotId)||0)+u.qty);

  }

  for(const [id,q] of used)

    if(ls.get(id).pending||q>ls.get(id).qty+1e-8)throw Error("Usage exceeds available stock.");

  for(const t of s.tasks)

    if(!text(t.title)||!t.title.trim()||!text(t.description)||!text(t.assignee)||

       !stages.includes(t.status)||!["Low","Medium","High"].includes(t.priority)||

       !validDate(t.due,true)||!text(t.productId)||(t.productId&&!ps.has(t.productId)))

      throw Error("Invalid workflow task.");

  return s;

}

try{

  raw=localStorage.getItem(KEY);

  if(raw){

    const d=JSON.parse(raw);

    state=validate(d.state);

    revision=d.revision||0;savedAt=d.savedAt||"";

  }

}catch(e){blocked=true;notify("Cannot load saved data. Original storage has not been overwritten. "+e.message);}


function commit(next){

  if(blocked)throw Error("Storage is unavailable or damaged. Export a recovery copy before continuing.");

  validate(next);

  if(localStorage.getItem(KEY)!==raw)throw Error("Another tab changed the data. Refresh before saving.");

  const d={version:1,state:next,revision:revision+1,savedAt:new Date().toISOString()};

  const nextRaw=JSON.stringify(d);

  if(raw)localStorage.setItem(PREVIOUS,raw);

  localStorage.setItem(KEY,nextRaw);

  state=next;revision=d.revision;savedAt=d.savedAt;raw=nextRaw;render();

}

const product=id=>state.products.find(p=>p.id===id);

const supplier=id=>state.suppliers.find(s=>s.id===id);

const order=id=>state.orders.find(o=>o.id===id);

const remaining=l=>l.pending?0:Math.max(0,Math.round((l.qty-state.usages.filter(u=>u.lotId===l.id).reduce((a,u)=>a+u.qty,0))*1e8)/1e8);

function stock(p){

  const lots=state.lots.filter(l=>l.productId===p.id);

  return {

    qty:lots.reduce((a,l)=>a+remaining(l),0),

    value:lots.reduce((a,l)=>a+remaining(l)*l.cost/l.qty,0),

    pending:lots.filter(l=>l.pending).reduce((a,l)=>a+l.qty,0)

  };

}

const btn=(a,id,label,cls="small")=>`<button class="${cls}" data-action="${a}" data-id="${esc(id)}">${esc(label)}</button>`;

const link=url=>url?`<a class="text-blue-600" href="${esc(safeURL(url))}" target="_blank" rel="noopener noreferrer">Open ↗</a>`:"—";

const table=(headers,rows)=>`<div class="bg-white border rounded-2xl overflow-auto"><table class="w-full"><thead><tr>${headers.map(h=>`<th>${h}</th>`).join("")}</tr></thead><tbody>${rows.length?rows.join(""):`<tr><td colspan="${headers.length}" class="text-center py-16 text-slate-400">No records yet. Add an item or import your backup.</td></tr>`}</tbody></table></div>`;


function render(){

  $("title").textContent=views[view][0];

  $("crumb").textContent=views[view][0];

  $("subtitle").textContent=views[view][1];

  $("sectionTitle").textContent=view==="overview"?"Workspace summary":views[view][0];

  $("search").classList.toggle("hidden",view==="overview");

  $("nav").innerHTML=Object.entries(views).map(([id,[title]])=>

    `<button data-view="${id}" class="rounded-xl px-4 py-3 text-sm text-left whitespace-nowrap ${view===id?"active":"text-slate-500"}">${esc(title)}</button>`).join("");

  $("actions").innerHTML=view==="workflows"?btn("task","","＋ New task","primary"):

    view==="suppliers"?btn("supplier","","＋ Add supplier","primary"):

    view==="usage"?btn("use","","＋ Record usage","primary"):

    btn("product","","＋ Add product","secondary")+" "+btn("receive","","＋ Purchase / receive","primary");


  const value=state.products.reduce((a,p)=>a+stock(p).value,0);

  const stats=[

    ["PRODUCTS",num(state.products.length),"Distinct products and variants"],

    ["STOCK VALUE",money(value),"Received stock still available"],

    ["INCOMING LOTS",num(state.lots.filter(l=>l.pending).length),"Not included in available stock"],

    ["OPEN TASKS",num(state.tasks.filter(t=>t.status!=="Done").length),"Across your workflow"]

  ];

  $("stats").innerHTML=stats.map(([a,b,c])=>`<div class="bg-white border rounded-2xl p-5"><p class="text-[10px] tracking-widest text-slate-400">${a}</p><p class="text-2xl font-bold mt-3">${b}</p><p class="muted mt-2">${c}</p></div>`).join("");

  const q=$("search").value.toLowerCase().trim(),match=(...v)=>v.join(" ").toLowerCase().includes(q);

  let rows=[];


  if(view==="overview"){

    const totals=stages.map(s=>[s,state.tasks.filter(t=>t.status===s).length]);

    const top=[...state.products].sort((a,b)=>stock(b).value-stock(a).value).slice(0,5);

    $("content").innerHTML=`

      <div class="grid xl:grid-cols-2 gap-5">

        <section class="bg-white border rounded-2xl p-6">

          <div class="flex justify-between mb-6"><h3 class="font-semibold">Workflow progress</h3><button data-view="workflows" class="text-sm text-blue-600">View board →</button></div>

          ${totals.map(([s,n])=>`<div class="mb-5"><div class="flex justify-between text-sm mb-2"><span>${s}</span><b>${n}</b></div><div class="h-2 rounded bg-slate-100"><div class="h-2 rounded bg-blue-500" style="width:${state.tasks.length?n/state.tasks.length*100:0}%"></div></div></div>`).join("")}

          <p class="muted">Workflow tasks organize work. Stock changes only through purchase, receipt, and usage actions.</p>

        </section>

        <section class="bg-white border rounded-2xl p-6">

          <h3 class="font-semibold mb-5">Highest stock value</h3>

          ${top.length?top.map(p=>`<div class="flex justify-between gap-3 py-4 border-b"><div><b class="text-sm">${esc(p.name)}</b><p class="muted">${esc(p.variant)}</p></div><span class="text-sm font-semibold">${money(stock(p).value)}</span></div>`).join(""):'<p class="text-slate-400 text-sm py-10">Import your inventory backup or add your first product.</p>'}

        </section>

      </div>`;

  }

  if(view==="inventory"){

    rows=state.products.filter(p=>match(p.name,p.variant,p.id,p.location)).map(p=>{

      const s=stock(p);

      return `<tr><td><b>${esc(p.name)}</b><p class="muted mt-1">${esc(p.variant)} · ${esc(p.id)}</p></td><td>${esc(p.location)||"—"}</td>

        <td><b class="${s.qty?"text-emerald-600":"text-slate-400"}">${num(s.qty)}</b> ${esc(p.unit)}${s.pending?`<p class="text-amber-600 text-xs mt-1">${num(s.pending)} incoming</p>`:""}</td>

        <td>${money(s.value)}</td><td><div class="flex gap-2">${btn("product",p.id,"Edit")}${btn("receive",p.id,"Buy")}${btn("use",p.id,"Use")}</div></td></tr>`;

    });

    $("content").innerHTML=table(["Product","Location","Available","Value","Actions"],rows);

  }

  if(view==="purchases"){

    rows=[...state.lots].reverse().filter(l=>{

      const p=product(l.productId),o=order(l.orderId);

      return match(p.name,p.variant,l.id,o.id,o.source,supplier(o.supplierId).name);

    }).map(l=>{

      const p=product(l.productId),o=order(l.orderId);

      return `<tr><td><b>${esc(p.name)}</b><p class="muted">${esc(p.variant)} · ${esc(l.id)}</p><p class="muted mt-1">${esc(o.source)}</p></td>

        <td>${esc(supplier(o.supplierId).name)}<div class="mt-1">${link(o.url)}</div></td>

        <td>${fmt(o.date)}<p class="muted">${esc(o.dateType)}</p></td>

        <td><span class="badge">${l.pending?"Incoming":"Received"}</span><p class="mt-2">${num(l.qty)} ${esc(p.unit)}</p><p class="muted">${num(remaining(l))} available</p></td>

        <td>${money(l.cost/l.qty)}</td><td>${money(l.cost)}<p class="muted">${l.estimated?"Estimated":"Confirmed"}</p></td>

        <td><div class="flex gap-2">${l.pending?btn("confirm",l.id,"Receive all"):""}${btn("cost",l.id,"Edit cost")}</div></td></tr>`;

    });

    $("content").innerHTML=table(["Product / lot","Supplier","Date","Quantity","Unit cost","Lot cost","Actions"],rows);

  }

  if(view==="usage"){

    rows=[...state.usages].reverse().filter(u=>{

      const l=state.lots.find(l=>l.id===u.lotId);

      return match(product(l.productId).name,u.person,u.note);

    }).map(u=>{

      const l=state.lots.find(l=>l.id===u.lotId),p=product(l.productId);

      return `<tr><td>${fmt(u.date)}</td><td><b>${esc(p.name)}</b><p class="muted">${esc(p.variant)}</p></td><td>${num(u.qty)} ${esc(p.unit)}</td><td>${money(u.qty*l.cost/l.qty)}</td><td>${esc(u.person)}</td><td>${esc(u.note)}</td></tr>`;

    });

    $("content").innerHTML=table(["Date","Product","Quantity","Cost","Used by","Purpose"],rows);

  }

  if(view==="suppliers"){

    rows=state.suppliers.filter(s=>match(s.name,s.url)).map(s=>`<tr><td><b>${esc(s.name)}</b></td><td>${link(s.url)}</td><td>${state.orders.filter(o=>o.supplierId===s.id).length}</td><td>${btn("supplier",s.id,"Edit")}</td></tr>`);

    $("content").innerHTML=table(["Supplier","Link","Orders","Actions"],rows);

  }

  if(view==="workflows"){

    $("content").innerHTML=`<div class="grid md:grid-cols-2 2xl:grid-cols-4 gap-4">${stages.map((stage,index)=>{

      const tasks=state.tasks.filter(t=>t.status===stage&&match(t.title,t.description,t.assignee,product(t.productId)?.name));

      return `<section class="bg-slate-100/80 border rounded-2xl p-3 min-h-64">

        <div class="flex justify-between items-center p-2 mb-3"><h3 class="text-sm font-semibold">${stage}</h3><span class="text-xs bg-white border rounded px-2 py-1">${tasks.length}</span></div>

        <div class="space-y-3">${tasks.map(t=>{

          const overdue=t.due&&t.due<today()&&t.status!=="Done";

          const color=t.priority==="High"?"text-red-600 bg-red-50":t.priority==="Medium"?"text-amber-700 bg-amber-50":"text-blue-600 bg-blue-50";

          return `<article class="bg-white border rounded-xl p-4 shadow-sm">

            <div class="flex justify-between items-center mb-3"><span class="text-[10px] px-2 py-1 rounded ${color}">${esc(t.priority)} priority</span>${btn("task",t.id,"Edit")}</div>

            <h4 class="font-semibold text-sm break-words">${esc(t.title)}</h4>

            <p class="text-xs text-slate-500 mt-2 whitespace-pre-wrap break-words">${esc(t.description)}</p>

            ${t.productId?`<p class="badge mt-3">${esc(product(t.productId).name)}</p>`:""}

            <div class="flex flex-wrap justify-between gap-2 text-xs mt-4 pt-3 border-t"><span>${esc(t.assignee)||"Unassigned"}</span><span class="${overdue?"text-red-600":"text-slate-400"}">${t.due?fmt(t.due):"No due date"}${overdue?" · Overdue":""}</span></div>

            <div class="flex justify-between mt-3">

              ${index?`<button class="small" data-move="${esc(t.id)}" data-stage="${stages[index-1]}">← Back</button>`:"<span></span>"}

              ${index<3?`<button class="small text-blue-600" data-move="${esc(t.id)}" data-stage="${stages[index+1]}">Next →</button>`:'<span class="text-xs text-emerald-600">✓ Complete</span>'}

            </div>

          </article>`;

        }).join("")||'<p class="text-xs text-slate-400 text-center py-8">No tasks in this stage</p>'}</div>

      </section>`;

    }).join("")}</div>`;

  }

  $("saved").textContent=savedAt?`Last saved ${new Date(savedAt).toLocaleString()} · Revision ${revision}`:"No data saved yet.";

}


function input(label,name,value="",type="text",required=true,full=false){

  return `<label class="${full?"sm:col-span-2":""}">${esc(label)}<input name="${name}" value="${esc(value)}" type="${type}" ${required?"required":""} ${type==="number"?'min="0" step="any"':'maxlength="2000"'}></label>`;

}

function select(label,name,options,value="",required=true){

  return `<label class="sm:col-span-2">${esc(label)}<select name="${name}" ${required?"required":""}><option value="">${required?"Select…":"None"}</option>${options.map(([id,title])=>`<option value="${esc(id)}" ${id===value?"selected":""}>${esc(title)}</option>`).join("")}</select></label>`;

}

function openForm(action,id=""){

  mode=action;editing=id;$("error").textContent="";

  let fields="",title="";

  if(action==="product"){

    const p=product(id)||{};

    title=id?"Edit product":"Add product";

$("form").addEventListener("submit",e=>{

  e.preventDefault();

  try{

    const f=Object.fromEntries(new FormData(e.target)),next=copy(state);

    const amount=(name,positive=false)=>{

      const v=Number(f[name]);

      if(f[name]===""||!Number.isFinite(v)||(positive?v<=0:v<0)||v>1e9)throw Error("Enter a valid quantity or cost.");

      return v;

    };

    if(mode==="product"){

      const p={id:editing||uid("P"),name:f.name.trim(),variant:f.variant.trim(),unit:f.unit.trim(),location:f.location.trim()};

      const i=next.products.findIndex(p=>p.id===editing);

      if(i>=0){

        if(next.lots.some(l=>l.productId===editing)&&p.unit!==next.products[i].unit)throw Error("Cannot change units after purchasing stock.");

        next.products[i]=p;

      }else next.products.push(p);

    }

    if(mode==="supplier"){

      const s={id:editing||uid("S"),name:f.name.trim(),url:safeURL(f.url.trim())};

      const i=next.suppliers.findIndex(s=>s.id===editing);

      if(i>=0)next.suppliers[i]=s;else next.suppliers.push(s);

    }

    if(mode==="receive"){

      const id=uid("O"),cost=amount("cost"),pending=f.status==="pending";

      next.orders.push({id,date:f.date,dateType:pending?"Recorded / incoming":"Received",supplierId:f.supplierId,total:cost,source:f.source.trim()||"Manual entry",url:safeURL(f.url.trim())});

      next.lots.push({id:uid("L"),productId:f.productId,orderId:id,qty:amount("qty",true),cost,estimated:false,pending});

    }

    if(mode==="use"){

      const l=state.lots.find(l=>l.id===f.lotId),qty=amount("qty",true);

      if(!l||qty>remaining(l)+1e-8)throw Error("Quantity exceeds available stock.");

      next.usages.push({id:uid("U"),lotId:l.id,qty,date:f.date,person:f.person.trim(),note:f.note.trim()});

    }

    if(mode==="confirm"){

      const l=next.lots.find(l=>l.id===editing);

      if(!l?.pending)throw Error("This lot is not awaiting receipt.");

      l.pending=false;

      const o=next.orders.find(o=>o.id===l.orderId);

      o.date=f.date;o.dateType="Received";

    }

    if(mode==="cost"){

      const l=next.lots.find(l=>l.id===editing);

      l.cost=amount("cost");l.estimated=false;

    }

    if(mode==="task"){

      const old=next.tasks.find(t=>t.id===editing);

      const t={id:editing||uid("T"),title:f.title.trim(),description:f.description.trim(),assignee:f.assignee.trim(),

        due:f.due,status:f.status,priority:f.priority,productId:f.productId,

        createdAt:old?.createdAt||new Date().toISOString(),updatedAt:new Date().toISOString()};

      const i=next.tasks.findIndex(t=>t.id===editing);

      if(i>=0)next.tasks[i]=t;else next.tasks.push(t);

    }

    commit(next);$("modal").close();notify("Saved successfully.");

  }catch(e){$("error").textContent=e.message;}

});

  const url=URL.createObjectURL(new Blob([text],{type:"application/json"}));

  const a=document.createElement("a");a.href=url;a.download=name;a.click();

}
